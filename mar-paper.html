extract_summary <- function(x) {
    sumx <- summary(x)
    a = round(sumx$Parameters['z',c('Estimate', '2.5%', '97.5%')], digits = 2)
    b = round(sumx$Parameters['z', 'Pr(>|t|)'], digits = 3)
    c = round(sumx$R2a, digits = 2)
    out <- c(a,b,c)
    return(out)
}
Mtypes = c('E', 'thetaw', 'thetapi')
marfit_list <- lapply(Mtypes, MARcalc, mardf = mardf)
summarydf <- do.call(rbind, lapply(marfit_list, extract_summary))
dimnames(summarydf)[[1]] <- Mtypes
dimnames(summarydf)[[2]][4:5] <- c('Pr(>|t|)', 'R2_adj')
print(summarydf)
